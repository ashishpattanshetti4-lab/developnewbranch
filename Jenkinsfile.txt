pipeline {
    agent any

    stages {
        stage('Pull Develop Branch') {
            steps {
                sh '''
                    mkdir -p /tmp/develop-branch-content
                    cp -r "$WORKSPACE"/*.txt /tmp/develop-branch-content/ 2>/dev/null || true
                    echo "Files copied to destination folder:"
                    ls -la /tmp/develop-branch-content
                '''
            }
        }
    }
}
